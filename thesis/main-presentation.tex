%% --- Bibliography -----------------------------------------------------------
%% Same references.bib as the thesis, but biblatex rather than the thesis's
%% natbib+plainnat: plainnat prints every author, and entries such as the
%% DeepSeek-R1 report have hundreds, which cannot fit on a slide. maxbibnames
%% truncates them to "et al.". backend=bibtex keeps the toolchain unchanged, so
%% build.sh's existing -bibtex flag still resolves the citations, and
%% natbib=true keeps the familiar \citep / \citet commands available.

%% --- Footline: JKU logo, running section, frame number, progress bar ---------

\setbeamertemplate{footline}{%
	\hbox{%
		%% The generous height and depth are deliberate. Beamer subtracts the whole
		%% footline box from \textheight, so "ht" reserves a blank strip between
		%% the lowest slide content and the footer text, and "dp" reserves one
		%% between the footer text and the bottom paper edge. Every content frame
		%% is then trimmed until the build reports no overfull \vbox, which is
		%% what actually guarantees the body never reaches into that strip.
		\begin{beamercolorbox}[wd=\paperwidth, ht=4.0ex, dp=3.2ex, leftskip=0.85cm, rightskip=0.85cm]{footline}%
			\usebeamerfont{footline}\color{jkugrey}%
			\raisebox{-0.55ex}{\includegraphics[height=1.55ex]{logos/jku_en_black_wide.pdf}}%
			\hspace{0.6em}\textbar\hspace{0.6em}%
			\insertsection%
			\hfill%
			\ifjkubackup Backup\else\insertframenumber\fi%
		\end{beamercolorbox}%
	}%
	%% Full-bleed talk-progress bar flush with the bottom paper edge. Dividing
	%% before multiplying keeps the intermediate below TeX's 2^30 sp ceiling,
	%% and the guard covers the first pass, when the total is still unknown.
	\setlength{\jkuprogress}{\paperwidth}%
	\ifjkubackup\else
		\ifnum\inserttotalframenumber>0\relax
			\divide\jkuprogress by \inserttotalframenumber%
			\multiply\jkuprogress by \insertframenumber%
		\else
			\setlength{\jkuprogress}{0pt}%
		\fi
	\fi
	\nointerlineskip%
	\begin{beamercolorbox}[wd=\paperwidth, ht=0.9pt, dp=0pt]{}%
		\color{jkurule}\rule{\paperwidth}{0.9pt}%
		\hspace*{-\paperwidth}\color{jkumagenta}\rule{\jkuprogress}{0.9pt}%
	\end{beamercolorbox}%
}

%% \takeaway{...} -- the single sentence the audience should retain, pinned to
%% the bottom of a content slide. Deliberately compact: on a 16:9 frame the body
%% area is only about 190pt tall, so this box may cost at most two lines.
\newcommand{\takeaway}[1]{%
	\vfill
	\setlength{\fboxsep}{4pt}%
	\noindent\colorbox{jkulight}{%
		\begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}%
			\footnotesize\raggedright\color{jkunavy}%
			\textcolor{jkumagenta}{\textbf{$\Rightarrow$}}~#1%
		\end{minipage}}%
	\vspace{1pt}%
}

%% \headline{...} -- a boxed statement used at the top of a slide. Ragged right
%% because a narrow column box justifies into visible inter-word rivers.
\newcommand{\headline}[1]{%
	\setlength{\fboxsep}{4pt}%
	\noindent\colorbox{jkulight}{%
		\begin{minipage}{\dimexpr\textwidth-2\fboxsep\relax}%
			\footnotesize\raggedright\color{jkunavy}#1%
		\end{minipage}}%
	\vspace{0.18cm}%
}

%% Lead-in label that opens a column.
\newcommand{\lead}[1]{{\color{jkunavy}\textbf{#1}}\par\vspace{1pt}}

\newcommand{\hl}[1]{\textcolor{jkumagenta}{\textbf{#1}}}
\newcommand{\nv}[1]{\textcolor{jkunavy}{\textbf{#1}}}

%% Numbers on slides are typeset a touch smaller than the surrounding prose so a
%% dense result line still reads as one sentence.
\newcommand{\num}[1]{\mbox{#1}}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Motivation \& Background}

\begin{frame}{The Credit-Assignment Gap in \acit{RLVR}}
	\headline{\acfit{RLVR} swaps the learned reward model for an \emph{external verifier}~\citep{deepseekmath, r1}: objective, but it speaks only once.}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{The price}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item \hl{One bit per trajectory}, at the very end.
				\item Spread over \hl{hundreds of tokens}: credit assignment falls to the policy gradient.
				\item Proxy rewards invite over-optimization~\citep{rewardoveropt, spuriousrewards}.
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{Existing ways to densify}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item \acpit{PRM}: label every step~\citep{verifystepbystep, mathshepherd} $-$ costs annotation or extra rollouts.
				\item Generative verifiers: a second model~\citep{genverifiers}.
				\item \nv{Self-rewarding}: the policy scores itself~\citep{selfrewardingllms, confasreward}.
			\end{itemize}
		\end{column}
	\end{columns}
	\takeaway{\textbf{Goal:} a \emph{bounded}, \emph{per-token} confidence signal, supervised by correctness, trained \emph{jointly} with the policy.}
\end{frame}

\begin{frame}{Closest Prior Work: \acit{LaSeR}}
	\framesubtitle{Last-Token Self-Rewarding~\citep{laser}}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{The mechanism}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item After the answer, read the log-probability of one control token $z_c$.
				\item Train it against the verifier bit with an \acit{MSE} loss, added to \acit{GRPO}.
			\end{itemize}
			\vspace{-0.15cm}
			\[
				\mathrm{score} = \beta_v \log \pi_\theta(z_c \mid x, y) - \beta_v c_{\mathrm{ref}}
			\]
			\vspace{-0.45cm}
			\begin{itemize}
				\item Verification costs \emph{one token}, not a 72B reward model.
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{Three costs it pays}
			\begin{enumerate}
				\setlength{\itemsep}{5pt}
				\item \hl{Sparse in time.} One score cannot say \emph{where} the derivation went wrong.
				\item \hl{Unbounded.} \citet{laser} report interference once the token takes too much mass.
				\item \hl{Extra forward pass.} The score lives \emph{after} \acit{EOS}, where decoding has stopped.
			\end{enumerate}
		\end{column}
	\end{columns}
	\takeaway{Each cost is a design choice, not a law $-$ so make the readout \emph{dense}, \emph{bounded} and \emph{in-band}.}
\end{frame}

\begin{frame}{Research Questions \& Contributions}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{Research questions}
			\begin{description}[RQ4]
				\setlength{\itemsep}{5pt}
				\item[\textcolor{jkumagenta}{RQ1}] Learn a \emph{per-token} confidence jointly with \acit{GRPO} without hurting accuracy?
				\item[\textcolor{jkumagenta}{RQ2}] Does \emph{bounding} the logit stop it leaking into the text?
				\item[\textcolor{jkumagenta}{RQ3}] Does feeding confidence back into the advantage help or hurt?
				\item[\textcolor{jkumagenta}{RQ4}] Can the signal steer \emph{inference-time} compute?
			\end{description}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{Contributions}
			\begin{enumerate}
				\setlength{\itemsep}{5pt}
				\item A \nv{modern \acit{LaSeR} reimplementation} on \texttt{trl}~\citep{trl} and \texttt{peft}~\citep{peft}.
				\item \nv{\textit{Dense Confidence}}: a bounded readout at \emph{every} response position.
				\item A \nv{controlled ablation grid} of six training-time variants on \acit{GSM8K}~\citep{gsm8k}.
				\item \nv{Three inference-time controllers}, no retraining.
			\end{enumerate}
		\end{column}
	\end{columns}
	\takeaway{To our knowledge the first system wiring a jointly trained \emph{dense} confidence head into all three test-time-compute mechanisms in one trainer.}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Method: \textit{Dense Confidence}}

\begin{frame}{\textit{Dense Confidence}: Readout and Objective}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{Read one token at every step}
			\begin{itemize}
				\setlength{\itemsep}{3pt}
				\item Repurpose \texttt{<|vision\_pad|>}, a token our text-only pipeline never inserts.
				\item A hook on \texttt{lm\_head}~\citep{pytorch} grabs its logit $-$ already computed, so \emph{no} extra pass.
				\item A scaled sigmoid bounds it:
			\end{itemize}
			\vspace{-0.3cm}
			\begin{equation*}
				c_t = \sigma\!\left(\tfrac{\pi}{\sigma_{\mathrm{conf}}\sqrt{3}}
				\bigl(\ell_t^{\mathrm{conf}} - \mu_{\mathrm{conf}}\bigr)\right) \in [0,1]
			\end{equation*}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{Supervise it, then optionally feed it back}
			\vspace{-0.1cm}
			\begin{equation*}
				L_{\mathrm{total}} = -\mathcal{J}_{\mathrm{GRPO}}(\theta) + \beta_{\mathrm{conf}} L_{\mathrm{conf}}
			\end{equation*}
			\vspace{-0.35cm}
			\begin{itemize}
				\setlength{\itemsep}{3pt}
				\item \nv{$+\ell$}: class-rebalanced \acit{BCE} against the binary reward, broadcast to \emph{all} $T_i$ positions.
				\item \nv{$+r$}: blend $\bar c_i$ into the advantage, $A_i = (1-\alpha)A_{r_i} + \alpha A_{\bar c_i}$ $-$ bounded, so no clipping schedule.
			\end{itemize}
		\end{column}
	\end{columns}
	\takeaway{\hl{$\beta_{\mathrm{conf}}$ is the fragile knob:} $10\times$ too large collapsed the policy onto \emph{always wrong, confidently so}.}
\end{frame}

\begin{frame}{Inference Time: Three Controllers, One Trained Head}
	\framesubtitle{Each targets a different test-time-compute knob~\citep{testtimecomputescaling}; none needs retraining}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.315\textwidth}
			\headline{\textbf{1. Answer selection}\\[1pt]\emph{knob: aggregation}}
			\small
			Over the $G$ rollouts with scores $s_i$:
			\begin{itemize}
				\item \acit{MV}: plain vote~\citep{selfconsistency}.
				\item \acit{HC}: $\arg\max_i s_i$.
				\item \acit{CW-MV}: vote weighted by $s_i$~\citep{deepthinkwithconfidence}.
			\end{itemize}
		\end{column}
		\begin{column}{0.315\textwidth}
			\headline{\textbf{2. Temperature modulation}\\[1pt]\emph{knob: sampling distribution}}
			\small
			Running mean $\bar c^{(H)}_t$ over $H\!=\!8$ steps, $w_t = 1-\bar c^{(H)}_t$:
			\vspace{-0.1cm}
			\[
				\tau_t = \tau_{\min} + w_t(\tau_{\max}-\tau_{\min})
			\]
			\vspace{-0.4cm}
			\begin{itemize}
				\item Exploit when confident, explore when not.
			\end{itemize}
		\end{column}
		\begin{column}{0.315\textwidth}
			\headline{\textbf{3. \acit{SMC} particle filtering}\\[1pt]\emph{knob: trajectory population}}
			\small
			The $G$ rollouts \emph{are} the particle set~\citep{smcsteering, twistedsmc}:
			\begin{itemize}
				\item Kill particles below $\delta_{\mathrm{filter}}$, clone donors $\propto \bar c^{(H)}$.
				\item Needs \texttt{DynamicCache} \acit{KV} surgery.
			\end{itemize}
		\end{column}
	\end{columns}
	\takeaway{Controllers \textbf{2} and \textbf{3} are \emph{structurally exclusive} to \textit{Dense}: a last-token \acit{LaSeR} score exposes no per-token potential.}
\end{frame}

\section{Assessment \& Conclusion}

\begin{frame}{Critical Assessment}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{\textcolor{jkunavy}{Strengths}}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item \nv{Minimal surface.} One token, one hook, one loss $-$ no second model, no step labels.
				\item \nv{Like-for-like comparison.} \acit{LaSeR} reimplemented \emph{inside the same trainer}.
				\item \nv{New capability.} Tempmod and \acit{SMC} are impossible for a last-token score.
				\item \nv{Reproducible.} Every number in this deck is read live from the released \texttt{CSVs}.
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{\textcolor{jkumagenta}{Weaknesses \& threats to validity}}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item \hl{One training seed.} The $\pm$ captures evaluation sampling, not the \acit{RL} seed lottery.
				\item \hl{No calibration metric.} \acit{MCC}/\acit{ROC-AUC} measure \emph{discrimination}; \acit{ECE} was never computed.
				\item \hl{Confounded.} \textit{Dense} vs.\ \acit{LaSeR} changes four things at once.
				\item \hl{Narrow regime.} One dataset, $0.5\,$B, $G=4$, no hyperparameter sweep.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Conclusion \& Outlook}
	\headline{A $0.5\,$B policy can be trained, in \emph{one} joint pass over $\calcpct{8799104}{502831872}{2}$ of its parameters, to reason \emph{and} to score its own correctness at every token.}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\lead{Established}
			\begin{itemize}
				\item \textit{Pass}@1 $\AccBeforePct{\AnsBase}{pass@1}{1} \to \AccAfterPct{\AnsDenseSR}{pass@1}{1}$, \acit{MCC} up to $\ConfAfterV{\ConfDenseAux}{mcc}{3}$, \acit{CW-MV} $\AccAfterPct{\AnsDenseAux}{confidence_weighted_majority_voting}{1}$.
				\item \nv{Not established:} calibration, seed robustness, a general win from the generative overlays.
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\lead{Outlook}
			\begin{itemize}
				\item Repeated seeds, and a joint sweep over $\beta_{\mathrm{conf}}$, $\alpha_{\mathrm{blend}}$, $\delta_{\mathrm{filter}}$, $G$.
				\item Move to code, where binary feedback is the real bottleneck.
				\item \nv{Mid-generation reversion:} roll back when confidence collapses $-$ a self-correcting decoder no last-token score can express.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Which Way Should Temperature Move?}
	\framesubtitle{The inversion ablation is a controlled sanity check on the direction of the effect}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.47\textwidth}
			\headline{\textbf{Default} $w_t = 1-\bar c^{(H)}_t$\\[1pt]\emph{Exploit when confident, explore when uncertain}}
			\lead{Math-exam analogue}
			\begin{itemize}
				\item On a practiced sub-problem, commit: $\tau \to \tau_{\min}$.
				\item On an uncertain one, a wrong guess costs nothing more, so \emph{explore}.
				\item The downside is \hl{bounded}.
			\end{itemize}
		\end{column}
		\begin{column}{0.49\textwidth}
			\headline{\textbf{Inverted} $w_t = \bar c^{(H)}_t$\\[1pt]\emph{Explore when confident, play safe when uncertain}}
			\lead{Race-car analogue}
			\begin{itemize}
				\item On a familiar corner, push off the safe line: $\tau \to \tau_{\max}$.
				\item On an unknown corner, ease off to avoid the crash.
				\item The downside is \hl{unbounded}.
			\end{itemize}
		\end{column}
	\end{columns}
	\takeaway{\acit{GSM8K} has the exam cost structure, so we \emph{predicted} the default mapping to win on classification $-$ and it does: \acit{MCC} $\num{\ConfAfter{\ConfDenseSR}{mcc}} \to \num{\ConfAfter{\ConfTempNoFil}{mcc}}$, vs.\ $\num{\ConfAfter{\ConfInvNoFil}{mcc}}$ inverted.}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Theoretical Connection to \acit{LaSeR}}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.50\textwidth}
			The KL-regularized optimum~\citep{rlhf} has the \acit{DPO} form~\citep{directpreferenceopt}. Inverting it for a \emph{confidence token} $z_c$ given $(x,y)$ yields the \emph{exact} reward
			\vspace{-0.1cm}
			\[
				r(x,y)=\beta_v\log\frac{\pi_\theta(z_c\mid x,y)}{\pi_{\mathrm{ref}}(z_c\mid x,y)}+\beta_v\log Z(x,y).
			\]
			\vspace{-0.35cm}
			\par Recovering the \acit{LaSeR} score needs \hl{two approximations}:
			\begin{itemize}
				\item $c_{\mathrm{ref}}\approx\log\pi_{\mathrm{ref}}(z_c\mid x,y)$ treated as \emph{constant};
				\item the partition function dropped, $Z(x,y)\approx 1$.
			\end{itemize}
		\end{column}
		\begin{column}{0.46\textwidth}
			\lead{Three orthogonal differences}
			\begin{enumerate}
				\setlength{\itemsep}{3pt}
				\item \nv{Signal density.} $T_i$ \acit{BCE} terms vs.\ one \acit{MSE} term.
				\item \nv{Bounded vs.\ unbounded.} $\bar c_i\in[0,1]$ makes the blend safe by construction.
				\item \nv{No post-\acit{EOS} pass.} The logit is read \emph{during} the regular forward.
			\end{enumerate}
		\end{column}
	\end{columns}
	\takeaway{\textbf{Honest caveat:} the experiment changes all three at once, so it compares \emph{configurations}, not signal density in isolation.}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Why \acit{SMC} Filtering Needs Cache Surgery}
	\begin{columns}[T, onlytextwidth]
		\begin{column}{0.50\textwidth}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item The $G$ rollouts of a \acit{GRPO} group \emph{are} the particle set, so the marginal compute cost is unchanged.
				\item At every step an \texttt{SMCLogitsProcessor} splits the active particles at $\delta_{\mathrm{filter}}$ into \emph{dead} and \emph{alive}.
				\item Each dead particle is replaced by an alive donor drawn $\propto \bar c^{(H)}$.
				\item Finished particles are masked out of \emph{both} sides, so a completed answer is never overwritten.
			\end{itemize}
		\end{column}
		\begin{column}{0.46\textwidth}
			\headline{The replacement is \emph{not} a relabeling of token streams: the donor's continuation depends on its full \acit{KV} history.}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item A pre-hook captures the \texttt{DynamicCache}; the donor's slice is cloned across \emph{all} layers.
				\item Copying the \emph{entire} prefix cache preserves causal correctness.
				\item Hence the overlay refuses to run under \texttt{vLLM}~\citep{vllm}.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
