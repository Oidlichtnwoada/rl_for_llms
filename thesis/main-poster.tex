\begin{columns}

	%% ===================== COLUMN 1 : PROBLEM AND IDEA =====================
	\column{0.30}

	\block{Motivation: reward is rich in value, poor in time}{%
		\begin{itemize}
			\item \acit{RLVR} gives an \emph{objective} reward from a verifier --- but only \textbf{one bit per
				      trajectory} (correct / incorrect).
			\item That single verdict must be spread across \emph{hundreds} of tokens $\Rightarrow$ hard credit
			      assignment.
			\item \textit{LaSeR} adds self-confidence, but only at the token \emph{after \acit{EOS}} (sparse) and as
			      an \emph{unbounded} log-probability that can leak into the generated text.
		\end{itemize}
		\textbf{Goal:} a \emph{bounded, per-token} confidence signal, calibrated to correctness and cheap enough
		to train \emph{jointly} with the policy.
	}

	\block{Research Questions}{%
		\begin{description}
			\item[\textcolor{jkuaccent}{RQ1}] Can a \emph{per-token} confidence signal be learned jointly with
			      \acit{GRPO} \emph{without} hurting reasoning accuracy?
			\item[\textcolor{jkuaccent}{RQ2}] Does \emph{bounding} the confidence logit stop it from leaking into
			      the text (the \textit{LaSeR} problem)?
			\item[\textcolor{jkuaccent}{RQ3}] Does feeding confidence back into training improve calibration of
			      correctness?
			\item[\textcolor{jkuaccent}{RQ4}] Can the signal be reused at \emph{inference} time to trade compute
			      for accuracy \emph{or} calibration?
		\end{description}
	}

	\block{Key terms}{%
		\begin{description}
			\item[\textcolor{jkuaccent}{\textit{Dense} vs.\ \textit{LaSeR}}] confidence read at \emph{every} token
			      (ours) vs.\ only from the token \emph{after the \acit{EOS} token} (baseline).
			\item[\textcolor{jkuaccent}{$+\ell{-}r$ / $+\ell{+}r$}] both train the auxiliary confidence loss
			      ($\ell$); the $+\ell{+}r$ variant \emph{additionally} feeds confidence into the \acit{GRPO}
			      advantage (\emph{self-rewarding}).
			\item[\textcolor{jkuaccent}{\acit{MV} / \acit{CW-MV}}] pick the final answer by a plain vs.\ a
			      \emph{confidence-weighted} majority vote of the $G$ samples.
			\item[\textcolor{jkuaccent}{\acit{MCC}}] \emph{Matthews Correlation Coefficient} --- how well
			      confidence predicts correctness: $0$ = random, $1$ = perfect.
		\end{description}
	}

	\block{Method: \textit{Dense Confidence} (ours)}{%
		\begin{itemize}
			\item Repurpose an \emph{unused} control token (\texttt{<|vision\_pad|>}); a hook reads its
			      logit $\ell_t^{\mathrm{conf}}$ at \emph{every} step and squashes it to a \emph{bounded}
			      confidence
			      $c_t = \sigma\!\big(\tfrac{\pi}{\sigma_{\mathrm{conf}}\sqrt{3}}(\ell_t^{\mathrm{conf}}-\mu_{\mathrm{conf}})\big)\in[0,1]$.
			\item Supervise it \emph{at every step} with a class-rebalanced \acit{BCE} loss \emph{added} to
			      \acit{GRPO} (it does \emph{not} flow through the policy gradient; tuning $\beta_{\mathrm{conf}}$ is
			      key for stability):
		\end{itemize}
		\vspace{-6pt}
		\[
			L_{\mathrm{total}} = -\mathcal{J}_{\mathrm{GRPO}}(\theta) + \beta_{\mathrm{conf}}\,L_{\mathrm{conf}},\quad
			L_{\mathrm{conf},i} = -\tfrac{1}{T_i}\!\sum_{t=1}^{T_i}\!\big[r_i\log c_{i,t} + (1{-}r_i)\log(1{-}c_{i,t})\big].
		\]
		\vspace{-10pt}
		\begin{itemize}
			\item \emph{Self-rewarding} ($+r$): blend the trajectory-mean confidence into the advantage.
			\item Shifting the sigmoid mean $\mu_{\mathrm{conf}}$ makes sampling the token \emph{arbitrarily
				      unlikely} $\Rightarrow$ \emph{no} leakage and \emph{no} extra forward pass.
		\end{itemize}
	}

	\block{Inference-time scaling --- reuse the head, no retraining}{%
		Three controllers reuse the trained \textit{Dense} head at evaluation only:
		\begin{itemize}
			\item \textbf{Answer selection:} vote over the $G$ samples by \acit{MV}, \acit{HC} (highest mean
			      confidence) or \acit{CW-MV}.
			\item \textbf{Temperature control:} raise/lower the decode temperature with a \emph{windowed}
			      running-mean confidence.
			\item \textbf{\acit{SMC} filtering:} kill low-confidence rollouts, clone high-confidence ones.
		\end{itemize}
		The last two \emph{need} a per-token signal --- only possible with our \textit{Dense} methodology,
		impossible for after-\acit{EOS} \textit{LaSeR}.
	}

		\begin{itemize}
			\item \textit{Dense} keeps a high \acit{MCC} ($>0.50$) and \acit{ROC-AUC}
			      $>0.84$ either way.
			\item The \emph{unbounded} \textit{LaSeR} reward \textbf{collapses} calibration once fed back into
			      training ($\ConfAfter{\ConfLaserAux}{mcc}\!\to\!\ConfAfter{\ConfLaserSR}{mcc}$).
			\item \textbf{No leakage (RQ2):} the confidence token is sampled
			      $\InclPctAfter{\AnsDenseSR}{0}$ of the time --- no extra forward pass.
			\item \textbf{Inference trade-off (RQ4):} on \emph{one} checkpoint, temperature $+$ \acit{SMC}
			      filtering pushes \acit{MCC} to $\ConfAfter{\ConfTempFil}{mcc}$ (calibration), while \acit{CW-MV}
			      maximizes answer accuracy.
		\end{itemize}
	}

	\block{Conclusion \& outlook}{%
		\begin{itemize}
			\item One repurposed token $+$ a hook $+$ a \acit{BCE} loss $=$ a calibrated, per-token confidence head
			      at $0.5\,$B parameters, \emph{for free}.
			\item Same head, two inference modes (no retraining): \acit{CW-MV} answer selection buys
			      \emph{accuracy}; temperature $+$ \acit{SMC} filtering buys \emph{calibration} (higher \acit{MCC}).
			\item Should scale with model size, data, and sample budget $G$; a natural \acit{PRM} distillation
			      target.
		\end{itemize}
